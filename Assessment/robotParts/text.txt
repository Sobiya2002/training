const allParts = [
  "Rosie_sensor", "optimus_speaker", "Rosie_horn", "Ruby_antenna",
  "victor_display", "Rosie_speaker", "Optimus_sensor", "Ruby_sensor",
  "Nova_claw", "Optimus_display", "Rosie_display", "Ruby_claw",
  "Optimus_claw", "Victor_antenna"
];
 
const required_parts1 = ["sensor", "speaker", "display", "claw"];
const required_parts2 = ["sensor", "speaker", "display"];
const required_parts3 = ["antenna", "claw"];
 
/**
* Build map: robot -> set(parts)
*/
function buildRobotPartsMap(allParts) {
  const map = {};
 
  allParts.forEach(item => {
    const [robot, part] = item.split("_");
    const robotKey = robot.toLowerCase(); // normalize name
 
    if (!map[robotKey]) map[robotKey] = new Set();//map lets you to group data in key and value like a dictionary where as set is used to sotre values without any duplicates
    map[robotKey].add(part.toLowerCase());// setting parts to the robot name.
  });
 
  return map;
}


function getRobotsWithParts(allParts, requiredParts) {
  const robotMap = buildRobotPartsMap(allParts); // {rosie: Set{'sensor','speaker','display'},...}
  const req = requiredParts.map(p => p.toLowerCase());// normalize required parts
 
  return Object.entries(robotMap) //change map to array [rosie: Set{'sensor','speaker','display'},...]
    .filter(([robot, parts]) => req.every(r => parts.has(r))) 
    .map(([robot]) => robot);
}
 
console.log("get_robot_for_required_part_1 =", getRobotsWithParts(allParts, required_parts1));
console.log("get_robot_for_required_part_2 =", getRobotsWithParts(allParts, required_parts2));
console.log("get_robot_for_required_part_3 =", getRobotsWithParts(allParts, required_parts3));
