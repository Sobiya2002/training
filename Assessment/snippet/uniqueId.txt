const get = async (u) => {

  const uniqueUserIds = get2(u);

  const result = [];

  for (const userId of uniqueUserIds) {

    const transactions = await connection.query(

      `SELECT * FROM transactions where userId = ${userId}`

    );

    let max = -1;

    transactions.forEach(transaction => {

      if (transaction.amount > max)

        max = transaction.amount;

    });

    result.push({ userId, max });

  }

  return result;

};

const get2 = (a) => {

  let result = [];

  for (let i = 0; i < a.length - 1; i++) {

    let isUnique = true;

    result.forEach(itemInResult => {

      if (itemInResult === a[i])

        isUnique = false;

    });

    if (isUnique) {

      result.push(a[i]);

    }

  }

  return result;

};
