return result; returns from the inner callback function (the one passed to conn.query)
It does not return from getStringFromDB

So getStringFromDB() returns immediately (with undefined) because it doesn’t wait for the database query to finish.
A return always returns from the function it is written in, and a callback is a different function that runs later, after the outer function has already finished.

Async/await doesn’t make a callback’s return “jump” to the outer function. What it does is:

it makes your outer function return a Promise, and
it lets you pause inside the outer function until that Promise is settled, and then return normally from the outer function.

That’s why the value becomes “reachable” to the caller.

What the code did

What went wrong
For the first word, start = 0
start - 1 becomes -1

Effect on execution
The reverse function accessed invalid indices
Characters became undefined or incorrectly swapped
Output was corrupted from the very first word

✅ Summary of Fixes + Defending Points

✅ FIX 1 — while (i < j) instead of while (i <= j)
❌ What was wrong
while (i <= j)

When i === j (middle of an odd‑length string), the code performs a redundant self‑swap
This does not change the value, but it is unnecessary work and makes reasoning about boundaries harder

✅ Why the fix is correct
while (i < j)

Each character pair is swapped exactly once
The middle character naturally stays in place
This is the standard two‑pointer termination condition used in string reversal algorithms [geeksforgeeks.org]

Interview defense:

Using i < j avoids redundant self‑swaps and clearly expresses that each symmetric pair is swapped once.


✅ FIX 2 — reverse() returns the full string
❌ What was wrong
return str.join("").substring(start, end);

Returned only a slice, not the full modified string
Caller expected a complete string, leading to corrupted concatenation

✅ Why the fix is correct
return arr.join("");

Returns the entire updated string
Matches the contract expected by reverseWords

Interview defense:

The bug was a function‑contract mismatch: callers expected a full string, but the function returned only a substring.

✅ Why the fix is correct

Last valid index is always length - 1
Prevents undefined values during swapping [geeksforgeeks.org]

Interview defense:

length is a count, not an index; valid indices go from 0 to length - 1.


✅ FIX 6 — Use s.length - 1 when reversing the entire string
❌ What was wrong
reverse(s, 0, s.length)

Same out‑of‑bounds issue
Final reversal was unreliable

✅ Why the fix is correct
reverse(s, 0, s.length - 1)

Ensures every swap is between valid indices
Correctly reverses the full string

Interview defense:

Reversing the entire string requires using the last valid index, not the length.

function reverse(str, start, end) {
  let temp;
  str = str.split("");
  var i = start;
  var j = end;

  // FIX: use i < j (avoid redundant self-swap when i === j)
  while (i < j) {
    temp = str[i];
    str[i] = str[j];
    str[j] = temp;
    i++;
    j--;
  }

  // FIX: return FULL modified string (not a substring)
  return str.join("");
}

// Function to reverse words (same logic preserved)
function reverseWords(s) {
  let start = 0;

  for (let end = 0; end < s.length; end++) {
    if (s[end] == ' ') {
      // FIX: correct bounds (reverse word [start, end-1], not start-1)
      s =
        s.substring(0, start) +
        reverse(s, start, end - 1) +
        s.substring(end);

      // FIX: next word starts after the space
      start = end + 1;
    }
  }

  // FIX: reverse the last word using last valid index s.length - 1
  s = s.substring(0, start) + reverse(s, start, s.length - 1);

  // FIX: reverse the entire string using last valid index s.length - 1
  s = reverse(s, 0, s.length - 1);

  return s;
}

const mysql = require("mysql");
const { reverseWords, getStringFromDB } = require("../index");

describe("reverseWords()", () => {
  test("reverses word order for normal sentence", () => {
    expect(reverseWords("Look there is a fox")).toBe("fox a is there Look");
  });

  test("handles two words", () => {
    expect(reverseWords("HI Bob")).toBe("Bob HI");
  });

  test("single word stays same", () => {
    expect(reverseWords("hello")).toBe("hello");
  });

  test("empty string returns empty", () => {
    expect(reverseWords("")).toBe("");
  });

  test("non-string input returns empty (DB defensive)", () => {
    expect(reverseWords(undefined)).toBe("");
    expect(reverseWords(null)).toBe("");
    expect(reverseWords(123)).toBe("");
  });

  test("multiple spaces preserved by algorithm", () => {
    // Your algorithm preserves spacing patterns between words
    expect(reverseWords("hello  world")).toBe("world  hello");
  });

  test("punctuation stays attached to words", () => {
    expect(reverseWords("hello, world!")).toBe("world! hello,");
  });
});

describe("getStringFromDB()", () => {
  let conn;

  beforeEach(() => {
    conn = {
      connect: jest.fn(),
      query: jest.fn(),
      end: jest.fn(),
    };
    mysql.createConnection.mockReturnValue(conn);
  });

  afterEach(() => {
    jest.clearAllMocks();
  });

  test("resolves with text when row exists", async () => {
    conn.connect.mockImplementation((cb) => cb(null));
    conn.query.mockImplementation((sql, params, cb) =>
      cb(null, [{ text: "Look there is a fox" }])
    );

    const result = await getStringFromDB(10);

    expect(result).toBe("Look there is a fox");
    expect(mysql.createConnection).toHaveBeenCalledTimes(1);

    // SQL injection-safe: placeholder + params array [3](https://stackoverflow.com/questions/71949351/how-to-use-named-parameters-in-node-mysql2-prepared-statements)[4](https://blogs.oracle.com/mysql/parameterizing-mysql-queries-in-node)
    expect(conn.query).toHaveBeenCalledWith(
      "SELECT text FROM sentences WHERE sentenceID = ?",
      [10],
      expect.any(Function)
    );

    expect(conn.end).toHaveBeenCalledTimes(1);
  });

  test("resolves with empty string when no rows", async () => {
    conn.connect.mockImplementation((cb) => cb(null));
    conn.query.mockImplementation((sql, params, cb) => cb(null, []));

    const result = await getStringFromDB(999);

    expect(result).toBe("");
    expect(conn.end).toHaveBeenCalledTimes(1);
  });

  test("resolves with empty string when row exists but missing text", async () => {
    conn.connect.mockImplementation((cb) => cb(null));
    conn.query.mockImplementation((sql, params, cb) => cb(null, [{}]));

    const result = await getStringFromDB(10);

    expect(result).toBe("");
    expect(conn.end).toHaveBeenCalledTimes(1);
  });

  test("rejects if connect fails and closes connection", async () => {
    conn.connect.mockImplementation((cb) => cb(new Error("connect failed")));

    await expect(getStringFromDB(10)).rejects.toThrow("connect failed");

    expect(conn.end).toHaveBeenCalledTimes(1);
    expect(conn.query).not.toHaveBeenCalled();
  });

  test("rejects if query fails and closes connection", async () => {
    conn.connect.mockImplementation((cb) => cb(null));
    conn.query.mockImplementation((sql, params, cb) =>
      cb(new Error("query failed"))
    );

    await expect(getStringFromDB(10)).rejects.toThrow("query failed");

    expect(conn.end).toHaveBeenCalledTimes(1);
  });

  test("is SQL-injection resilient (input treated as value, not SQL)", async () => {
    conn.connect.mockImplementation((cb) => cb(null));
    conn.query.mockImplementation((sql, params, cb) =>
      cb(null, [{ text: "Safe" }])
    );

    const injection = "10 OR 1=1";
    const result = await getStringFromDB(injection);

    expect(result).toBe("Safe");
    expect(conn.query).toHaveBeenCalledWith(
      "SELECT text FROM sentences WHERE sentenceID = ?",
      [injection],
      expect.any(Function)
    );
  });
